option(MALEMOD_SURFACE_PROCESS_ISOLATED "One session per process with source parallel geometry" OFF)

option(MALEMOD_SURFACE_SERIAL_DIAGNOSTIC "Diagnostic process replay with serial geometry" OFF)
if(MALEMOD_BUILD_SURFACE_RUNTIME)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Source surface parity is verified for 32-bit only; configure -A Win32. The 64-bit parity gate currently fails.")
  endif()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  find_package(Threads REQUIRED)
  set(MALEMOD_EIGEN_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/legacy/wolverine/third-party" CACHE PATH "SDK-free Eigen include directory")
  if(NOT EXISTS "${MALEMOD_EIGEN_INCLUDE}/Eigen/SparseCholesky")
    message(FATAL_ERROR "The surface runtime needs Eigen, not a game or graphics SDK")
  endif()
  set(surface_generated "${CMAKE_CURRENT_BINARY_DIR}/surface-generated/surface-kernel.inc")
  set(surface_isolation_args "")
  if(MALEMOD_SURFACE_PROCESS_ISOLATED)
    if(NOT MSVC)
      message(FATAL_ERROR "Source parallel replay currently requires the verified MSVC scheduler")
    endif()
    set(surface_isolation_args "--process-isolated")
    if(MALEMOD_SURFACE_SERIAL_DIAGNOSTIC)
      if(NOT MALEMOD_ALLOW_UNVERIFIED_PROCESS_RUNTIME)
        message(FATAL_ERROR "Serial process experiments require explicit diagnostic opt-in; deploy the verified parallel recipe.")
      endif()
      list(APPEND surface_isolation_args "--serial-geometry")
    endif()
  endif()
  file(GLOB surface_sources CONFIGURE_DEPENDS "legacy/wolverine/src/runtime/*.h" "legacy/wolverine/src/runtime/*.cpp")
  add_custom_command(OUTPUT "${surface_generated}"
    COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tools/extract_surface_runtime.py" --output "${surface_generated}" ${surface_isolation_args}
    DEPENDS tools/extract_surface_runtime.py tools/data/surface-closure.json ${surface_sources}
    VERBATIM)
  add_library(malemod_surface STATIC src/surface/runtime.cpp "${surface_generated}")
  if(MALEMOD_COMPLIANT_SCROTAL_SKIN)
    target_compile_definitions(malemod_surface PRIVATE MALEMOD_COMPLIANT_SCROTAL_SKIN=1)
  endif()
  target_link_libraries(malemod_surface PUBLIC malemod Threads::Threads)
  if(MALEMOD_COMPLIANT_SCROTAL_SKIN)
    add_executable(scrotal_surface_test tests/scrotal_surface_test.cpp)
    target_link_libraries(scrotal_surface_test PRIVATE malemod_surface)
    set_property(TARGET scrotal_surface_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    add_executable(relative_dynamics_test tests/relative_dynamics_test.cpp)
    target_link_libraries(relative_dynamics_test PRIVATE malemod_surface)
    set_property(TARGET relative_dynamics_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    set(MALEMOD_DYNAMICS_REFERENCE_LIBRARY "" CACHE FILEPATH "Optional historical library for matched response comparison")
    if(MALEMOD_DYNAMICS_REFERENCE_LIBRARY)
      add_executable(relative_dynamics_reference_test tests/relative_dynamics_test.cpp)
      target_link_libraries(relative_dynamics_reference_test PRIVATE malemod "${MALEMOD_DYNAMICS_REFERENCE_LIBRARY}")
      set_property(TARGET relative_dynamics_reference_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    endif()
  endif()
  target_include_directories(malemod_surface PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/surface-generated" "${MALEMOD_EIGEN_INCLUDE}")
  if(MSVC)
    if(MALEMOD_SURFACE_PROCESS_ISOLATED)
      # Source-global storage uses the original runtime's arithmetic. The TLS
      # recipe is independently verified with strict arithmetic.
      target_compile_options(malemod_surface PRIVATE /fp:precise /utf-8)
    else()
      target_compile_options(malemod_surface PRIVATE /fp:strict /utf-8)
    endif()
  else()
    target_compile_options(malemod_surface PRIVATE -fno-fast-math -ffp-contract=off -msse2)
  endif()
  if(MALEMOD_SURFACE_PROCESS_ISOLATED)
    target_compile_definitions(malemod_surface PRIVATE MALEMOD_SURFACE_PROCESS_ISOLATED=1)
  endif()
  add_executable(surface_runtime_cli tests/surface_runtime_cli.cpp)
  target_link_libraries(surface_runtime_cli PRIVATE malemod_surface)
  add_executable(garment_support_runtime_cli tests/garment_support_runtime_cli.cpp)
  target_link_libraries(garment_support_runtime_cli PRIVATE malemod_surface)
  if(MALEMOD_SURFACE_PROCESS_ISOLATED)
    add_executable(surface_session_test tests/surface_process_test.cpp)
  else()
    add_executable(surface_session_test tests/surface_session_test.cpp)
  endif()
  target_link_libraries(surface_session_test PRIVATE malemod_surface)
  if(NOT MALEMOD_SURFACE_PROCESS_ISOLATED)
    add_executable(surface_collision_test tests/surface_collision_test.cpp)
    target_link_libraries(surface_collision_test PRIVATE malemod_surface)
    if(MSVC)
      set_property(TARGET surface_collision_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
  endif()
  if(MSVC)
    set_property(TARGET malemod_surface surface_runtime_cli garment_support_runtime_cli surface_session_test
      PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
endif()
