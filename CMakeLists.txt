cmake_minimum_required(VERSION 3.16)
project(MaleModBase VERSION 0.1.0 LANGUAGES CXX)
add_library(malemod INTERFACE)
target_include_directories(malemod INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(malemod INTERFACE cxx_std_17)
include(CTest)
if(BUILD_TESTING)
  add_executable(surface_limit_test tests/surface_limit_test.cpp)
  target_link_libraries(surface_limit_test PRIVATE malemod)
  add_test(NAME surface_limit COMMAND surface_limit_test)
endif()
