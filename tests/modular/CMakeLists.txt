cmake_minimum_required(VERSION 3.16)
project(MaleModModularRegression LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
get_filename_component(base "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
foreach(name modular_motion modular_transfer raphe_profile suspension_hang suspension_profile skin_relief convex_surface_cut surface_stitch compound_graft)
  add_executable(${name}_test "${base}/tests/${name}_test.cpp")
  target_include_directories(${name}_test PRIVATE "${base}/include")
  target_compile_features(${name}_test PRIVATE cxx_std_17)
  target_link_libraries(${name}_test PRIVATE Threads::Threads)
  if(MSVC)
    set_property(TARGET ${name}_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
  endif()
  add_test(NAME ${name} COMMAND ${name}_test)
endforeach()
add_executable(audit_surface_intersections "${base}/tools/audit_surface_intersections.cpp")
target_compile_features(audit_surface_intersections PRIVATE cxx_std_17)
add_test(NAME surface_intersection_fixtures COMMAND audit_surface_intersections --self-test)
