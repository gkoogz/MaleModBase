cmake_minimum_required(VERSION 3.16)
project(MaleModModularRegression LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
get_filename_component(base "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
foreach(name modular_motion modular_transfer raphe_profile suspension_hang suspension_skin_frame suspension_profile skin_relief convex_surface_cut surface_stitch compound_graft compound_contact live_controls)
  add_executable(${name}_test "${base}/tests/${name}_test.cpp")
  target_include_directories(${name}_test PRIVATE "${base}/include")
  target_compile_features(${name}_test PRIVATE cxx_std_17)
  target_link_libraries(${name}_test PRIVATE Threads::Threads)
  if(MSVC)
    set_property(TARGET ${name}_test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
  endif()
  add_test(NAME ${name} COMMAND ${name}_test)
endforeach()
add_executable(audit_surface_intersections "${base}/tools/audit_surface_intersections.cpp")
target_compile_features(audit_surface_intersections PRIVATE cxx_std_17)
add_test(NAME surface_intersection_fixtures COMMAND audit_surface_intersections --self-test)
add_executable(evaluate_compound_cut "${base}/tools/evaluate_compound_cut.cpp")
target_include_directories(evaluate_compound_cut PRIVATE "${base}/include")
target_compile_features(evaluate_compound_cut PRIVATE cxx_std_17)
add_executable(evaluate_compound_contact "${base}/tools/evaluate_compound_contact.cpp")
target_include_directories(evaluate_compound_contact PRIVATE "${base}/include")
target_compile_features(evaluate_compound_contact PRIVATE cxx_std_17)
add_executable(evaluate_compound_dynamic "${base}/tools/evaluate_compound_dynamic.cpp")
target_include_directories(evaluate_compound_dynamic PRIVATE "${base}/include")
target_compile_features(evaluate_compound_dynamic PRIVATE cxx_std_17)
target_link_libraries(evaluate_compound_dynamic PRIVATE Threads::Threads)
